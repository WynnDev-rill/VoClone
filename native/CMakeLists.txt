cmake_minimum_required(VERSION 3.22)
project(VoCloneJNI LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
foreach(runtime omni whisper)
    file(GLOB_RECURSE archives "${CMAKE_CURRENT_SOURCE_DIR}/build/${runtime}-${RUNTIME_ABI}/*.a")
    add_library(vc_${runtime} SHARED ${runtime}.cpp)
    target_include_directories(vc_${runtime} PRIVATE "build/sources/omnivoice/src" "build/sources/whisper/include" "build/sources/whisper/ggml/include")
    target_link_libraries(vc_${runtime} PRIVATE -Wl,--start-group ${archives} -Wl,--end-group android log)
    target_compile_options(vc_${runtime} PRIVATE -fvisibility=hidden)
    target_link_options(vc_${runtime} PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic -Wl,-z,max-page-size=16384)
endforeach()
file(GLOB LAME_SOURCES CONFIGURE_DEPENDS "mp3/Lame/libmp3lame/*.c")
add_library(vc_mp3 SHARED mp3.c mp3/Mp3Encoder.c ${LAME_SOURCES})
target_include_directories(vc_mp3 PRIVATE mp3/include mp3/Lame mp3/Lame/include)
target_compile_definitions(vc_mp3 PRIVATE HAVE_CONFIG_H)
target_compile_options(vc_mp3 PRIVATE -fvisibility=hidden)
target_link_libraries(vc_mp3 PRIVATE m)
target_link_options(vc_mp3 PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic -Wl,-z,max-page-size=16384)
