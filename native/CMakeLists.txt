cmake_minimum_required(VERSION 3.22)
project(VoCloneJNI LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(runtimes omni whisper)
if(RUNTIME_ABI STREQUAL "arm64-v8a")
    list(APPEND runtimes omni_i8mm)
endif()
foreach(runtime ${runtimes})
    set(archive_dir "${runtime}-${RUNTIME_ABI}")
    set(source "${runtime}.cpp")
    if(runtime STREQUAL "omni_i8mm")
        set(archive_dir "omni-i8mm-${RUNTIME_ABI}")
        set(source omni.cpp)
    endif()
    file(GLOB_RECURSE archives "${CMAKE_CURRENT_SOURCE_DIR}/build/${archive_dir}/*.a")
    add_library(vc_${runtime} SHARED ${source})
    if(runtime STREQUAL "whisper")
        target_include_directories(vc_${runtime} PRIVATE "build/sources/whisper/include" "build/sources/whisper/ggml/include")
    else()
        target_include_directories(vc_${runtime} PRIVATE "build/sources/omnivoice/src" "build/sources/ggml/include" "build/sources/ggml/src")
    endif()
    if(runtime STREQUAL "omni_i8mm")
        target_compile_definitions(vc_${runtime} PRIVATE VC_I8MM_VARIANT)
    endif()
    target_link_libraries(vc_${runtime} PRIVATE -Wl,--start-group ${archives} -Wl,--end-group android log)
    target_compile_options(vc_${runtime} PRIVATE -fvisibility=hidden)
    target_link_options(vc_${runtime} PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic -Wl,-z,max-page-size=16384)
endforeach()
file(GLOB LAME_SOURCES CONFIGURE_DEPENDS "mp3/Lame/libmp3lame/*.c")
add_library(vc_mp3 SHARED mp3.c mp3/Mp3Encoder.c ${LAME_SOURCES})
target_include_directories(vc_mp3 PRIVATE mp3/include mp3/Lame mp3/Lame/include)
target_compile_definitions(vc_mp3 PRIVATE HAVE_CONFIG_H)
target_compile_options(vc_mp3 PRIVATE -fvisibility=hidden)
target_link_libraries(vc_mp3 PRIVATE m)
target_link_options(vc_mp3 PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic -Wl,-z,max-page-size=16384)
